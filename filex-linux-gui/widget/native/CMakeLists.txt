cmake_minimum_required(VERSION 3.21)
project(FilexWidget LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Qml Quick)
qt_policy(SET QTP0001 NEW)
qt_add_qml_module(filexwidget URI Filex.Widget VERSION 1.0
    OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../plasmoid/contents/native"
    SOURCES directory.h directory.cpp)
target_link_libraries(filexwidget PRIVATE Qt6::Quick)
set_target_properties(filexwidget PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../plasmoid/contents/native" BUILD_RPATH "$ORIGIN")
set_target_properties(filexwidgetplugin PROPERTIES BUILD_RPATH "$ORIGIN")
include(CTest)
if(BUILD_TESTING)
    add_executable(widget-probe probe.cpp)
    target_link_libraries(widget-probe PRIVATE filexwidget Qt6::Quick)
    add_test(NAME widget-probe COMMAND widget-probe)
endif()
option(FILEX_WIDGET_HOST_TESTS "Build isolated Plasma host probe (needs Plasma development headers)" OFF)
if(BUILD_TESTING AND FILEX_WIDGET_HOST_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    find_library(FILEX_PLASMA Plasma REQUIRED)
    find_library(FILEX_PLASMAQUICK PlasmaQuick REQUIRED)
    add_executable(widget-host-probe host-probe.cpp)
    target_link_libraries(widget-host-probe PRIVATE Qt6::Quick Qt6::Widgets ${FILEX_PLASMA} ${FILEX_PLASMAQUICK})
    target_include_directories(widget-host-probe PRIVATE /usr/include/Plasma /usr/include/PlasmaQuick /usr/include/KF6/KConfigCore /usr/include/KF6/KConfig /usr/include/KF6/KCoreAddons /usr/include/KF6/KPackage)
endif()
