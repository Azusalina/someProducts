cmake_minimum_required(VERSION 3.20)
project(ToolkitDesktop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
find_package(KF6GlobalAccel REQUIRED)
find_package(KF6WindowSystem REQUIRED)
add_executable(toolkit-hotkeys hotkeys.cpp)
target_link_libraries(toolkit-hotkeys PRIVATE Qt6::Widgets Qt6::DBus KF6::GlobalAccel)
add_library(toolkit-glass SHARED glass.cpp)
target_link_libraries(toolkit-glass PRIVATE Qt6::Widgets KF6::WindowSystem)
option(TOOLKIT_BUILD_GESTURE "Build the native KWin modifier-click effect" ON)
if(TOOLKIT_BUILD_GESTURE)
    find_package(KWin REQUIRED)
    add_library(toolkitgesture MODULE gesture.cpp)
    target_link_libraries(toolkitgesture PRIVATE KWin::kwin)
endif()
